   543→async function createAuditorium(theatreId: string, name: string, layout: LayoutConfig, priceOverrides?: Record<string, number>) {
   544→  const { seats } = generateLayout(layout);
   545→  const aud = await db.auditorium.create({
   546→    data: {
   547→      theatreId, name, layoutConfig: JSON.stringify(layout), geometrySource: "PROCEDURAL", status: "PUBLISHED",
   548→      categories: {
   549→        create: Object.entries(DEFAULT_PRICES).map(([code, v]) => ({
   550→          code, name: v.name, color: v.color, basePrice: priceOverrides?.[code] ?? v.price,
   551→        })),
   552→      },
   553→      models: {
   554→        create: {
   555→          kind: "PROCEDURAL", status: "PUBLISHED", version: 1,
   556→          notes: "Procedural demonstration geometry generated from the auditorium layout configuration. Not a calibrated survey of a real auditorium.",
   557→        },
   558→      },
   559→    },
   560→    include: { categories: true },
   561→  });
   562→  const catByCode = new Map(aud.categories.map((c) => [c.code, c.id]));
   563→  await db.seat.createMany({
   564→    data: seats.map((s) => ({
   565→      auditoriumId: aud.id, rowLabel: s.rowLabel, seatNumber: s.seatNumber, rowIndex: s.rowIndex, colIndex: s.colIndex,
   566→      x: s.x, z: s.z, baseY: s.baseY, eyeHeight: s.eyeHeight, categoryId: catByCode.get(s.categoryCode)!,
   567→      isAccessible: s.isAccessible,
   568→    })),
   569→  });
   570→  return { id: aud.id, seatCount: seats.length };
   571→}
